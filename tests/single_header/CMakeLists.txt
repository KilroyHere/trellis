# The unit tests again, against the single header (plan §10, T1.8; label single_header).
#
# tools/amalgamate.py writes the single header into this build directory (not into
# single_include/, so that builds of several presets don't share one file), and a stub for each
# public and internal header, <trellis/X.hpp>, includes it: the unit test sources compile unchanged,
# and every Trellis declaration they see comes from the single header. Each unit test program of
# tests/unit/ that links trellis::trellis gets a twin, trellis_single_header_<name>, with the same
# sources, definitions, options, include directories and libraries, but the stubs in place of
# trellis::trellis and include/. The twins follow tests/unit/CMakeLists.txt as it grows.
set(single_header "${CMAKE_CURRENT_BINARY_DIR}/single_include/trellis.hpp")
file(GLOB_RECURSE trellis_headers CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/include/trellis/*.hpp")
add_custom_command(OUTPUT "${single_header}"
  COMMAND "${Python3_EXECUTABLE}" -I "${PROJECT_SOURCE_DIR}/tools/amalgamate.py"
          --output "${single_header}"
  DEPENDS "${PROJECT_SOURCE_DIR}/tools/amalgamate.py" ${trellis_headers}
  COMMENT "Generating the single header"
  VERBATIM)
add_custom_target(trellis_single_header DEPENDS "${single_header}")

set(stub_dir "${CMAKE_CURRENT_BINARY_DIR}/include")
foreach(header IN LISTS trellis_headers)
  file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}/include" "${header}")
  file(CONFIGURE OUTPUT "${stub_dir}/${relative}" @ONLY CONTENT [[
// Generated by tests/single_header/CMakeLists.txt: <@relative@> for the tests that run against
// the single header, which provides every Trellis header.
#include "@single_header@"
]])
endforeach()
add_library(trellis_single_header_stubs INTERFACE)
target_include_directories(trellis_single_header_stubs INTERFACE "${stub_dir}")

# fail_on_include_dir(TARGET PROPERTY): stops the configuration when PROPERTY of TARGET, a unit test
# program, names include/. Its include directories come before the stubs on a twin's command line,
# so include/ there would hand the twin the real headers, silently, instead of the single header.
set(include_dir "${PROJECT_SOURCE_DIR}/include")
cmake_path(NORMAL_PATH include_dir)
function(fail_on_include_dir target property)
  get_target_property(entries ${target} ${property})
  get_target_property(source_dir ${target} SOURCE_DIR)
  if(NOT entries)
    return()
  endif()
  foreach(entry IN LISTS entries)
    string(REGEX REPLACE "^-I" "" path "${entry}")
    if(NOT path MATCHES "^\\$<")  # a generator expression is searched as text below
      cmake_path(ABSOLUTE_PATH path BASE_DIRECTORY "${source_dir}" NORMALIZE)
      string(REGEX REPLACE "/+$" "" path "${path}")
    endif()
    string(FIND "${entry}" "${include_dir}" position)
    if(path STREQUAL include_dir OR (entry MATCHES "^\\$<" AND position GREATER -1))
      message(FATAL_ERROR "tests/single_header: ${target}'s ${property} names include/ "
        "(${entry}), so its single-header twin would test the headers instead of the single "
        "header. Take include/ from trellis::trellis, which the twin replaces.")
    endif()
  endforeach()
endfunction()

get_property(unit_targets DIRECTORY "${PROJECT_SOURCE_DIR}/tests/unit" PROPERTY BUILDSYSTEM_TARGETS)
# The programs link Threads::Threads, an imported target that find_package makes visible only in
# the directory that calls it (tests/unit/), so this one finds it too.
find_package(Threads REQUIRED)
set(twins "")
foreach(original IN LISTS unit_targets)
  get_target_property(type ${original} TYPE)
  get_target_property(libraries ${original} LINK_LIBRARIES)
  if(NOT type STREQUAL "EXECUTABLE"
     OR NOT ("trellis::trellis" IN_LIST libraries OR "trellis" IN_LIST libraries))
    continue()
  endif()
  list(REMOVE_ITEM libraries trellis::trellis trellis)
  fail_on_include_dir(${original} INCLUDE_DIRECTORIES)
  fail_on_include_dir(${original} COMPILE_OPTIONS)
  get_target_property(source_dir ${original} SOURCE_DIR)
  get_target_property(sources ${original} SOURCES)
  set(absolute_sources "")
  foreach(source IN LISTS sources)
    cmake_path(ABSOLUTE_PATH source BASE_DIRECTORY "${source_dir}")
    list(APPEND absolute_sources "${source}")
  endforeach()

  string(REGEX REPLACE "^trellis_" "" name "${original}")
  set(twin trellis_single_header_${name})
  add_executable(${twin} ${absolute_sources})
  target_link_libraries(${twin} PRIVATE trellis_single_header_stubs ${libraries})
  foreach(property IN ITEMS COMPILE_DEFINITIONS COMPILE_OPTIONS INCLUDE_DIRECTORIES CXX_STANDARD)
    get_target_property(value ${original} ${property})
    if(value)
      set_property(TARGET ${twin} PROPERTY ${property} "${value}")
    endif()
  endforeach()
  add_dependencies(${twin} trellis_single_header)
  # Out of compile_commands.json, so that clang-tidy and clangd see each source with include/, as
  # its own program builds it: through a stub, include-cleaner would call every include unused.
  set_target_properties(${twin} PROPERTIES EXPORT_COMPILE_COMMANDS OFF)

  if("GTest::gmock_main" IN_LIST libraries OR "GTest::gtest_main" IN_LIST libraries)
    gtest_discover_tests(${twin}
      DISCOVERY_MODE PRE_TEST TEST_PREFIX single_header.${name}. PROPERTIES LABELS single_header)
  else()
    add_test(NAME single_header.${name} COMMAND ${twin})
    set_tests_properties(single_header.${name} PROPERTIES LABELS single_header)
  endif()
  list(APPEND twins ${twin})
endforeach()
if(NOT twins)
  message(FATAL_ERROR
    "tests/single_header: no unit test program of tests/unit links trellis::trellis")
endif()

# The single header compiled as the main file of a translation unit, as a precompiled header or a
# header unit compiles it, with the project warnings as errors, on GCC and Clang: nothing in it may
# depend on being included, such as a system_header pragma, which GCC reports in a main file with
# no flag to switch the warning off (T1.15; config.hpp, TRELLIS_MODULE_SYSTEM_HEADER). It needs the
# generated header, which the twins' build makes.
if(NOT MSVC)
  separate_arguments(trellis_cxx_flags UNIX_COMMAND "${CMAKE_CXX_FLAGS}")
  # The timetrace preset's -ftime-trace profiles code generation, so -fsyntax-only leaves it
  # unused, which Clang reports and the warnings-as-errors flag makes fatal. The test is about the
  # header's own warnings, so the profiler flag goes.
  list(FILTER trellis_cxx_flags EXCLUDE REGEX "^-ftime-trace")
  add_test(NAME single_header.main_file
    COMMAND "${CMAKE_CXX_COMPILER}" ${trellis_cxx_flags} -std=c++20 -fsyntax-only -x c++-header
            ${TRELLIS_WARNING_FLAGS} ${TRELLIS_WARNINGS_AS_ERRORS} "${single_header}")
  set_tests_properties(single_header.main_file PROPERTIES LABELS single_header)
endif()
