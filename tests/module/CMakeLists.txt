# Smoke tests of the C++20 modules trellis and trellis.testing (plan §5.2, §10, R15; label module),
# registered when TRELLIS_ENABLE_MODULE is on: a program whose translation units import the
# modules, with one that includes the headers instead; the unit tests that use only public names,
# through the modules; and the check that the modules export every public name.
#
# An importer must be compiled with the language options of the module it reads, so the module is
# built here with the tests' settings (C++20 without extensions) and, like every target of the
# project, with the warning set as errors.
set_target_properties(trellis_module PROPERTIES
  CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(trellis_module PRIVATE
  ${TRELLIS_WARNING_FLAGS} ${TRELLIS_WARNINGS_AS_ERRORS})

add_executable(trellis_module_tests
  import_test.cpp config_test.cpp headers_test.cpp testing_test.cpp)
trellis_module_importer(trellis_module_tests)
target_link_libraries(trellis_module_tests PRIVATE GTest::gmock_main)
gtest_discover_tests(trellis_module_tests
  DISCOVERY_MODE PRE_TEST TEST_PREFIX module. PROPERTIES LABELS module)

# The unit tests that use only public names, again through the modules (module.<program>.*): for
# each unit test program of tests/unit/ whose build the compiled modules can serve, its sources
# that never name detail are compiled with a stub for every header, <trellis/X.hpp>, that includes
# <trellis/config.hpp> for the macros and imports trellis (trellis.testing for the testing
# headers). A source that names detail can't run this way, since no module exports it, and neither
# can a program built in another language mode or with other options (the C++23 and C++26 programs,
# those without exceptions), nor one that sets a configuration macro, such as
# TRELLIS_HAS_ATOMIC_WAIT=0 for trellis_wait_fallback_tests: a macro reaches the module only from
# its own build. The selection is read from the sources, so CMake reconfigures when one changes.
set(import_stub_dir "${CMAKE_CURRENT_BINARY_DIR}/import_include")
file(GLOB_RECURSE trellis_headers CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/include/trellis/*.hpp")
foreach(header IN LISTS trellis_headers)
  file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}/include" "${header}")
  set(module_name trellis)
  if(relative MATCHES "^trellis/testing/")
    set(module_name trellis.testing)
  endif()
  file(CONFIGURE OUTPUT "${import_stub_dir}/${relative}" @ONLY CONTENT [[
// Generated by tests/module/CMakeLists.txt: <@relative@> for the unit tests that run through the
// C++20 modules. A module carries no macros, so config.hpp provides them.
#include "@PROJECT_SOURCE_DIR@/include/trellis/config.hpp"
import @module_name@;
]])
endforeach()

get_property(unit_targets DIRECTORY "${PROJECT_SOURCE_DIR}/tests/unit" PROPERTY BUILDSYSTEM_TARGETS)
# The programs link Threads::Threads, an imported target that find_package makes visible only in
# the directory that calls it (tests/unit/), so this one finds it too.
find_package(Threads REQUIRED)
# A target starts with its directory's options (the project warnings); what it adds is its own.
get_property(directory_options DIRECTORY "${PROJECT_SOURCE_DIR}/tests/unit" PROPERTY COMPILE_OPTIONS)
get_target_property(standard_definitions trellis_unit_tests COMPILE_DEFINITIONS)
set(module_programs "")
foreach(original IN LISTS unit_targets)
  get_target_property(type ${original} TYPE)
  get_target_property(libraries ${original} LINK_LIBRARIES)
  get_target_property(standard ${original} CXX_STANDARD)
  get_target_property(options ${original} COMPILE_OPTIONS)
  get_target_property(definitions ${original} COMPILE_DEFINITIONS)
  if(options)
    list(REMOVE_ITEM options ${directory_options})
  endif()
  if(NOT type STREQUAL "EXECUTABLE" OR NOT "GTest::gmock_main" IN_LIST libraries
     OR (standard AND NOT standard EQUAL 20) OR options
     OR NOT definitions STREQUAL standard_definitions)
    continue()
  endif()
  get_target_property(sources ${original} SOURCES)
  get_target_property(source_dir ${original} SOURCE_DIR)
  set(public_sources "")
  set(prelude_includes "")
  foreach(source IN LISTS sources)
    cmake_path(ABSOLUTE_PATH source BASE_DIRECTORY "${source_dir}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" text)
    # detail::name, or detail at the end of a qualified name: namespace d = trellis::detail;
    if(text MATCHES "detail[ \t]*::" OR text MATCHES "::[ \t]*detail([^A-Za-z0-9_]|$)")
      continue()
    endif()
    list(APPEND public_sources "${source}")
    # A unit test includes the Trellis headers first, which here imports a module before its
    # standard and GoogleTest headers. Clang 20 with libstdc++ and Clang 18 with libc++ reject
    # some standard headers included after an import of a module whose global module fragment
    # included them too ("'lifetimebound' attribute cannot be applied", "redefinition of concept
    # '__is_derived_from_optional'"), so a prelude, included before each source, includes the
    # sources' other headers first, the order delivery.md recommends to users.
    file(STRINGS "${source}" include_lines REGEX "^[ \t]*#[ \t]*include[ \t]*<")
    foreach(line IN LISTS include_lines)
      string(REGEX MATCH "<[^>]+>" header_name "${line}")
      if(NOT header_name MATCHES "^<trellis/")
        list(APPEND prelude_includes "${header_name}")
      endif()
    endforeach()
  endforeach()
  if(NOT public_sources)
    continue()
  endif()
  string(REGEX REPLACE "^trellis_" "" name "${original}")
  set(program trellis_module_${name})
  list(REMOVE_DUPLICATES prelude_includes)
  list(SORT prelude_includes)
  set(prelude "${CMAKE_CURRENT_BINARY_DIR}/${program}_prelude.hpp")
  set(prelude_text "// Generated by tests/module/CMakeLists.txt: the headers that ${original}\n")
  string(APPEND prelude_text "// includes, before the import that the stubs make.\n")
  foreach(header_name IN LISTS prelude_includes)
    string(APPEND prelude_text "#include ${header_name}\n")
  endforeach()
  file(CONFIGURE OUTPUT "${prelude}" CONTENT "${prelude_text}")

  add_executable(${program} ${public_sources})
  trellis_module_importer(${program})
  get_target_property(include_dirs ${original} INCLUDE_DIRECTORIES)
  if(include_dirs)
    target_include_directories(${program} PRIVATE ${include_dirs})
  endif()
  # BEFORE: the stubs precede include/, which trellis::module carries for config.hpp.
  target_include_directories(${program} BEFORE PRIVATE "${import_stub_dir}")
  if(MSVC)
    target_compile_options(${program} PRIVATE "/FI${prelude}")
  else()
    target_compile_options(${program} PRIVATE -include "${prelude}")
  endif()
  list(REMOVE_ITEM libraries trellis::trellis trellis)
  target_link_libraries(${program} PRIVATE ${libraries})
  target_compile_definitions(${program} PRIVATE ${definitions})
  # Out of compile_commands.json for the reason given in tests/single_header/CMakeLists.txt.
  set_target_properties(${program} PROPERTIES EXPORT_COMPILE_COMMANDS OFF)
  gtest_discover_tests(${program}
    DISCOVERY_MODE PRE_TEST TEST_PREFIX module.${name}. PROPERTIES LABELS module)
  list(APPEND module_programs ${program})
endforeach()
if(NOT module_programs)
  message(FATAL_ERROR "tests/module: no unit test program can run through the module")
endif()

# The export lists: each module of modules/ exports exactly the public names of its headers, which
# tools/module_exports.py reads from Clang's syntax tree, so only Clang builds run it. Its
# self-test plants missing, extra and misplaced exports in temporary trees.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU")
  add_test(NAME module.exports
    COMMAND "${Python3_EXECUTABLE}" -I "${PROJECT_SOURCE_DIR}/tools/module_exports.py"
            --config "${TRELLIS_COMPILER_CONFIG}")
  add_test(NAME tools.module_exports_selftest
    COMMAND "${Python3_EXECUTABLE}" -I
            "${PROJECT_SOURCE_DIR}/tests/tools/module_exports_selftest.py"
            --config "${TRELLIS_COMPILER_CONFIG}")
  set_tests_properties(module.exports tools.module_exports_selftest PROPERTIES
    LABELS "module;tools")
endif()
